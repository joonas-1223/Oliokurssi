# Raportin sijainti

# Vaihda tähän oman repositoriosi osoite
GITHUB_URL="https://github.com/joonas-1223/Linux_administration"

# Funktio levytilan tilan tarkistamiseen
check_disk_status() {
    local usage="$1"

    if (( usage >= 80 )); then
        echo "WARNING"
    else
        echo "OK"
    fi
}

# Levytilan varoituslogiikan testaaminen
if [[ "${1:-}" == "--test-status" ]]; then
    for usage in 50 80 90; do
        echo "$usage%: $(check_disk_status "$usage")"
    done
    exit 0
fi

# Kerätään palvelimen tiedot
GENERATED_AT=$(date '+%Y-%m-%d %H:%M:%S %Z %z')
UPTIME_TEXT=$(uptime -p)

# Levyn käyttöprosentti
DISK_USAGE=$(df -P / | awk 'NR == 2 {print $5}')
DISK_PERCENT=${DISK_USAGE%\%}

# Levytilan tila
DISK_STATUS=$(check_disk_status "$DISK_PERCENT")

# RAM-muistin määrä mebitavuina
MEM_TOTAL=$(free -m | awk '/^Mem:/ {print $2}')
MEM_AVAILABLE=$(free -m | awk '/^Mem:/ {print $7}')

# Raportin hakemisto ja väliaikainen tiedosto
mkdir -p "$OUTPUT_DIR"
TEMP_FILE=$(mktemp "$OUTPUT_DIR/.report.XXXXXX")

# Poista väliaikainen tiedosto virhetilanteessa
trap 'rm -f "$TEMP_FILE"' EXIT
